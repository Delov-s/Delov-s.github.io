## 3.解题思路

从 `APK` 中提取 `assets/init_res`，按`native`解密逻辑还原`DEX`，运行`exp`还原出真正的`dex`。

`DEX`解密：

```
#!/usr/bin/env python3

import hashlib
import struct
import sys
import zlib
from pathlib import Path


def ror8(value: int, count: int) -> int:
    """8 位循环右移"""
    value &= 0xFF
    return ((value >> count) | (value << (8 - count))) & 0xFF


def decrypt_init_res(encrypted: bytes, package_name: str) -> bytes:
    """
    根据 native 中的解密逻辑还原 DEX。
    """
    package_bytes = package_name.encode("utf-8")

    # 生成 16 字节循环密钥
    key = bytes(
        package_bytes[i % len(package_bytes)] ^ 0x5A
        for i in range(16)
    )

    decrypted = bytearray(len(encrypted))
    previous = 0xAA

    for i, current in enumerate(encrypted):
        value = previous ^ i ^ current ^ key[i & 0x0F]
        decrypted[i] = ror8(value, 2)

        # 注意：保存的是上一个密文字节
        previous = current

    return bytes(decrypted)


def fix_dex_checksums(data: bytes) -> bytes:
    """
    修复 DEX 的 SHA-1 和 Adler32。
    """
    data = bytearray(data)

    if data[:4] != b"dex\n":
        raise ValueError(
            "解密结果不是 DEX 文件，头部为: "
            + data[:16].hex(" ")
        )

    if len(data) < 0x70:
        raise ValueError("DEX 文件长度异常")

    # SHA-1 覆盖 offset 0x20 到文件末尾
    sha1_digest = hashlib.sha1(data[32:]).digest()
    data[12:32] = sha1_digest

    # Adler32 覆盖 offset 0x0C 到文件末尾
    adler32_value = zlib.adler32(data[12:]) & 0xFFFFFFFF
    struct.pack_into("<I", data, 8, adler32_value)

    return bytes(data)


def main():
    if len(sys.argv) != 4:
        print(
            "用法:\n"
            f"  {sys.argv[0]} <加密文件> <包名> <输出dex>\n\n"
            "示例:\n"
            f"  {sys.argv[0]} assets/init_res "
            "com.xiaoyuyc.mobile2 real.dex"
        )
        sys.exit(1)

    input_file = Path(sys.argv[1])
    package_name = sys.argv[2]
    output_file = Path(sys.argv[3])

    if not input_file.is_file():
        print(f"错误：文件不存在: {input_file}")
        sys.exit(1)

    encrypted = input_file.read_bytes()

    try:
        decrypted = decrypt_init_res(encrypted, package_name)

        if not decrypted.startswith(b"dex\n"):
            print("解密失败：输出不是 DEX")
            print("输出头部:", decrypted[:16].hex(" "))
            sys.exit(2)

        fixed_dex = fix_dex_checksums(decrypted)
        output_file.write_bytes(fixed_dex)

    except Exception as exc:
        print(f"处理失败: {exc}")
        sys.exit(2)

    print(f"解密成功: {output_file}")
    print(f"文件大小: {len(fixed_dex)} bytes")
    print(f"DEX magic: {fixed_dex[:8]!r}")


if __name__ == "__main__":
    main()
```

def rol8(value, count):
    return ((value << count) | (value >> (8 - count))) & 0xff


cipher = [
    0xC9, 0x88, 0x65, 0x91, 0xCB, 0xB3, 0x08, 0x50,
    0xA9, 0x21, 0x88, 0x90, 0x39, 0x5A, 0x6D, 0x17,
    0x59, 0xAE, 0xF4, 0xC5, 0x61, 0xD7, 0x86, 0x44,
    0x51, 0x5C
]

state = 0x5A
result = []

for i, target in enumerate(cipher):
    for ch in range(0x20, 0x7F):
        value = (rol8(state ^ ch, 3) + 3 * i) & 0xFF
        if value == target:
            result.append(chr(ch))
            state = value
            break

print("".join(result))
